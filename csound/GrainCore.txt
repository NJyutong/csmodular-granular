; Original CsModular granular engine. No granular synthesis opcodes.
; A private 64-voice pool renders one control block at a time with audio-rate
; vector operations: a grain costs a few dozen opcode calls per block instead of per
; sample, an idle engine costs almost nothing, and results stay identical to a
; per-sample renderer (bit-identical at ksmps=1; the only difference at larger ksmps
; is that the voice-count gain smoother is updated once per block).
; Grain starts stay sample-accurate: a grain born at offset j of a block has age -j,
; and its envelope is clamped so every sample before its start or after its end is silent.
; mode: 0 = sample, 1 = live ring, 2 = periodic waveform (220 Hz root).
opcode CMPulvisOutputView, 0, ai
    aSignal, iDisplay xin
    iScope = iDisplay + 1
    iSpectrum = iDisplay + 2
    aPhase phasor sr / 4096
    tablew aSignal, aPhase * 4096, iScope
    kHead downsamp aPhase
    tablew (round(kHead * 4096) + ksmps) % 4096, 4096, iScope
    fOutput pvsanal aSignal, 2048, 512, 2048, 1
    kNewSpectrum pvsftw fOutput, iSpectrum
endop

; Both signal-rate voices and the display evaluate the same window definition.
#define CM_HANN(P) # (0.5 - 0.5 * cos(6.283185307179586 * $P)) #
#define CM_TRIANGLE(P) # (1 - abs(2 * $P - 1)) #
#define CM_BLACKMAN(P) # (0.42 - 0.5 * cos(6.283185307179586 * $P) + 0.08 * cos(12.566370614359172 * $P)) #

opcode CMGrainCoreExtended, aa, aikkkkkkkkkkkkkkkkkkkkko
    aInput, iMode, kTable, kFrames, kSourceRate, kDensity, kSizeMs, kPosition, kPosJitter, kPitch, kPitchJitter, kWindow, kPan, kSpread, kFreeze, kSeed, kTimeJitter, kReverse, kFeedback, kSnapshotSelect, kSnapshotA, kSnapshotB, kCapture, iDisplay xin
    iBlock = ksmps
    ; Exact 0..ksmps-1 sample offsets of the current block.
    iRampTable ftgentmp 0, 0, -iBlock, -2, 0
    iFill = 0
    while iFill < iBlock do
        tableiw iFill, iFill, iRampTable
        iFill = iFill + 1
    od
    ; Live ring: the whole block is written before any grain reads it, so grains keep
    ; (ksmps - 4) extra samples away from the oldest data. Zero at the host's ksmps = 4.
    iMargin = max(0, iBlock - 4)

    kLength[] init 64
    kSource[] init 64
    kList[] init 64
    kAge[] init 64
    kSpan[] init 64
    kRead[] init 64
    kSeamWidth[] init 64
    kSeamValue[] init 64
    kStep[] init 64
    kWarpScale[] init 64
    kWarpRadians[] init 64
    kSourceLength[] init 64
    kShape[] init 64
    kLeft[] init 64
    kRight[] init 64
    ; A block may contain the tail of an expiring voice and a replacement in the
    ; same slot. Keep render snapshots separate from the reusable persistent pool.
    ; 128 holds the 64-slot pool plus up to 64 sample-accurate replacements
    ; inside the supported control blocks (the host uses ksmps=4).
    kBlockPoolSlot[] init 128
    kBlockSource[] init 128
    kBlockSourceLength[] init 128
    kBlockAge[] init 128
    kBlockSpan[] init 128
    kBlockReadIndex[] init 128
    kBlockStep[] init 128
    kBlockLength[] init 128
    kBlockSeamWidth[] init 128
    kBlockSeamValue[] init 128
    kBlockWarpScale[] init 128
    kBlockWarpRadians[] init 128
    kBlockShape[] init 128
    kBlockLeft[] init 128
    kBlockRight[] init 128
    kNextTrigger init -1
    kPreviousTimeOffset init 0
    kCarrier init 0
    ; Display-only peak history in physical ring order. Capture it alongside the audio.
    kRingPeak[] init 192
    kCapturedPeak[] init 192
    kLastPeakBin init -1
    kWrite init 0
    kFilled init 0
    kRandom init 1
    kTimeRandom init 1
    kLastSeed init -1
    kHeldPosition init 0
    kLastFreeze init -1
    kLastCapture init 0
    kSnapshotActive init 0
    kSnapshotWrite init 0
    kSnapshotFilled init 0
    kFeedbackPrevious init 0
    kCount init 0
    kSmoothedActive init 1

    aRamp tablera iRampTable, 0, 0

    kFrames = max(2, int(kFrames))
    kDensity limit kDensity, 0, 200
    kSizeMs limit kSizeMs, 2, 500
    kPosition limit kPosition, 0, 1
    kPosJitter limit kPosJitter, 0, 1
    kPitch limit kPitch, -48, 48
    kPitchJitter limit kPitchJitter, 0, 24
    kPan limit kPan, 0, 1
    kSpread limit kSpread, 0, 1
    kTimeJitter limit kTimeJitter, 0, 1
    kReverse = (kReverse >= 0.5 ? 1 : 0)
    kFeedback limit kFeedback, 0, 0.95
    kSnapshotSelect = (kSnapshotSelect >= 0.5 ? 1 : 0)
    kSeedValue = 1 + int(limit(kSeed, 0, 65534))
    if kSeedValue != kLastSeed then
        kRandom = kSeedValue
        kTimeRandom = (kSeedValue * 48271 + 12820163) % 2147483647
        if kTimeRandom == 0 then
            kTimeRandom = 1
        endif
        kLastSeed = kSeedValue
    endif
    if kFreeze < 0.5 || kLastFreeze < 0 then
        kHeldPosition = kPosition
    endif
    kLastFreeze = kFreeze

    ; Only the live source writes; freezing never suspends existing voices.
    ; The block is written first; each grain start sees the writer position of its own sample.
    if iMode == 1 && kFreeze < 0.5 then
        aWriteIndex = kWrite + aRamp
        if kFeedback > 0 then
            aRecord = aInput + kFeedback * kFeedbackPrevious
            aRecord limit aRecord, -1, 1
        else
            aRecord = aInput
        endif
        tablewkt aRecord, aWriteIndex, kTable, 0, 0, 1
        if iDisplay > 0 then
            kBin = int(kWrite * 192 / kFrames)
            kPeak max_k aRecord, 1, 1
            if kBin != kLastPeakBin then
                kRingPeak[kBin] = 0
                kLastPeakBin = kBin
            endif
            kRingPeak[kBin] = max(kRingPeak[kBin], kPeak)
        endif
    endif

    ; Capture to the inactive private table. Do not overwrite a table still read by
    ; an active grain or a tail queued for this block. tablecopy runs in Csound and
    ; publishes the new pointer only after the complete table has been copied.
    kCaptureEdge = (kCapture >= 0.5 && kLastCapture < 0.5 ? 1 : 0)
    kLastCapture = kCapture
    if iMode == 1 && kCaptureEdge == 1 && kFilled > 0 then
        kDestination = (kSnapshotActive == 1 ? kSnapshotB : kSnapshotA)
        kDestinationBusy = 0
        kCheckVoice = 0
        while kCheckVoice < 64 do
            if kLength[kCheckVoice] > 0 && kSource[kCheckVoice] == kDestination then
                kDestinationBusy = 1
            endif
            kCheckVoice = kCheckVoice + 1
        od
        if kDestinationBusy == 0 && kDestination >= 1 then
            tablecopy kDestination, kTable
            if iDisplay > 0 then
                kCapturedPeak = kRingPeak
            endif
            kSnapshotActive = (kDestination == kSnapshotA ? 1 : 2)
            if kFreeze < 0.5 then
                kSnapshotWrite = (kWrite + iBlock) % kFrames
                kSnapshotFilled = min(kFilled + iBlock, kFrames)
            else
                kSnapshotWrite = kWrite
                kSnapshotFilled = kFilled
            endif
        endif
    endif

    kReadTable = kTable
    kReadWrite = kWrite
    kReadFilled = kFilled
    if iMode == 1 && kSnapshotSelect == 1 then
        if kSnapshotActive == 0 then
            kDensity = 0
        elseif kSnapshotActive == 1 then
            kReadTable = kSnapshotA
            kReadWrite = kSnapshotWrite
            kReadFilled = kSnapshotFilled
        else
            kReadTable = kSnapshotB
            kReadWrite = kSnapshotWrite
            kReadFilled = kSnapshotFilled
        endif
    endif

    ; Only voices that can expire inside this block need a snapshot: their slot may
    ; be reused before the old tail is rendered. Other voices stay in the pool.
    kBlockCount = 0
    kItem = 0
    while kItem < kCount do
        kVoice = kList[kItem]
        if kLength[kVoice] > 0 then
            if kAge[kVoice] + iBlock >= kLength[kVoice] then
                kBlockPoolSlot[kBlockCount] = -1
                kBlockSource[kBlockCount] = kSource[kVoice]
                kBlockSourceLength[kBlockCount] = kSourceLength[kVoice]
                kBlockAge[kBlockCount] = kAge[kVoice]
                kBlockSpan[kBlockCount] = kSpan[kVoice]
                kBlockReadIndex[kBlockCount] = kRead[kVoice]
                kBlockStep[kBlockCount] = kStep[kVoice]
                kBlockLength[kBlockCount] = kLength[kVoice]
                kBlockSeamWidth[kBlockCount] = kSeamWidth[kVoice]
                kBlockSeamValue[kBlockCount] = kSeamValue[kVoice]
                kBlockWarpScale[kBlockCount] = kWarpScale[kVoice]
                kBlockWarpRadians[kBlockCount] = kWarpRadians[kVoice]
                kBlockShape[kBlockCount] = kShape[kVoice]
                kBlockLeft[kBlockCount] = kLeft[kVoice]
                kBlockRight[kBlockCount] = kRight[kVoice]
            else
                kBlockPoolSlot[kBlockCount] = kVoice
            endif
            kBlockCount = kBlockCount + 1
        endif
        kItem = kItem + 1
    od

    ; Sample-countdown scheduling preserves average density. A separate seeded stream
    ; offsets event times by at most 45% of the nominal period without reordering them.
    if iMode == 2 then
        kCarrierStep = pow(2, kPitch / 12) * kSourceRate / sr
    endif
    kOffset = 0
    while kOffset < iBlock do
        kShouldTrigger = 0
        if kDensity <= 0 then
            kNextTrigger = -1
            kPreviousTimeOffset = 0
        else
            if kNextTrigger < 0 then
                kNextTrigger = sr / kDensity
            endif
            kNextTrigger = kNextTrigger - 1
            if kNextTrigger <= 0 then
                kShouldTrigger = 1
            endif
        endif
        if kShouldTrigger == 1 then
            kSlot = 0
            kFound = 0
            while kSlot < 64 && kFound == 0 do
                ; The per-sample reference releases a voice before the first
                ; sample whose age reaches its length. Match that boundary even
                ; when it falls inside this control block.
                if kLength[kSlot] == 0 || kAge[kSlot] + kOffset >= kLength[kSlot] then
                    kLength[kSlot] = 0
                    kFound = 1
                else
                    kSlot = kSlot + 1
                endif
            od
            if kFound == 1 then
                ; Park-Miller state belongs to this UDO instance, not Csound globally.
                kRandom = (kRandom * 16807) % 2147483647
                kPositionRandom = 2 * kRandom / 2147483647 - 1
                kRandom = (kRandom * 16807) % 2147483647
                kPitchRandom = 2 * kRandom / 2147483647 - 1
                kRandom = (kRandom * 16807) % 2147483647
                kPanRandom = 2 * kRandom / 2147483647 - 1
                kRatio = pow(2, limit(kPitch + kPitchRandom * kPitchJitter, -48, 48) / 12)
                kRate = kRatio * kSourceRate / sr
                kStepRate = (kReverse == 1 ? -kRate : kRate)
                kDuration = max(2, int(kSizeMs * sr / 1000))
                if iMode == 2 then
                    ; Warp and wavetable start are independent at grain creation.
                    kWarpScale[kSlot] = kHeldPosition * kFrames / 6.283185307179586
                    kWarpRadians[kSlot] = 6.283185307179586 / kFrames
                else
                    kStart = limit(kHeldPosition + kPositionRandom * kPosJitter, 0, 1)
                endif
                if iMode == 1 then
                    ; Writer state exactly as it is at this sample.
                    if kSnapshotSelect == 1 then
                        kWriteNow = kReadWrite
                        kFilledNow = kReadFilled
                    elseif kFreeze < 0.5 then
                        kWriteNow = (kWrite + kOffset + 1) % kFrames
                        kFilledNow = min(kFilled + kOffset + 1, kFrames)
                    else
                        kWriteNow = kWrite
                        kFilledNow = kFilled
                    endif
                    ; Reserve the whole read trajectory AND future writer trajectory.
                    ; This remains safe if Freeze is released during a grain.
                    kDuration = min(kDuration, int((kFilledNow - 4 - iMargin) / (kRate + 1)))
                    if kReverse == 1 then
                        kNear = 2
                        kFar = kFilledNow - kDuration * (kRate + 1) - 2 - iMargin
                        kStart = kWriteNow - (kNear + kStart * max(0, kFar - kNear))
                    else
                        kNear = kDuration * kRate + 2
                        kFar = kFilledNow - kDuration - 2 - iMargin
                        kStart = kWriteNow - (kNear + kStart * max(0, kFar - kNear))
                    endif
                    kStepRate = (kReverse == 1 ? -kRate : kRate)
                elseif iMode == 2 then
                    ; Align overlapping grains to the carrier. Scatter phase symmetrically around it.
                    kStart = kCarrier + kCarrierStep * kOffset + 0.5 * kFrames * kPositionRandom * kPosJitter
                    kStepRate = (kReverse == 1 ? -kRate : kRate)
                else
                    kStart = kStart * (kFrames - 1)
                    kStepRate = (kReverse == 1 ? -kRate : kRate)
                endif
                if kDuration >= 2 then
                    kVoicePan = limit(kPan + 0.5 * kSpread * kPanRandom, 0, 1)
                    ; Read position the grain would have at sample 0 of this block.
                    kBlockRead = kStart - kStepRate * kOffset
                    kLength[kSlot] = kDuration
                    kAge[kSlot] = -kOffset
                    kSpan[kSlot] = kDuration - 1
                    kRead[kSlot] = kBlockRead - floor(kBlockRead / kFrames) * kFrames
                    kStep[kSlot] = kStepRate
                    kSource[kSlot] = (iMode == 1 ? kReadTable : kTable)
                    kSourceLength[kSlot] = kFrames
                    kSeamWidth[kSlot] = 0
                    if iMode == 0 && kFrames >= 16 && kStart + kRate * (kDuration - 1) > kFrames - 1 then
                        kHead tablekt 0, kTable
                        kTail tablekt kFrames - 1, kTable
                        if abs(kHead - kTail) > 0.001 then
                            ; Smooth approximately 2 ms on each side, with a bound for short files.
                            kSeamWidth[kSlot] = min(kFrames * 0.125, max(2, 0.002 * sr * kRate))
                            kSeamValue[kSlot] = (kHead + kTail) * 0.5
                        endif
                    endif
                    kShape[kSlot] = int(limit(kWindow, 0, 2) + 0.5)
                    kLeft[kSlot] = sqrt(1 - kVoicePan)
                    kRight[kSlot] = sqrt(kVoicePan)
                    ; Keep the persistent active-slot list ordered without rescanning all 64 slots each block.
                    kItem = 0
                    kFound = 0
                    while kItem < kCount && kFound == 0 do
                        if kList[kItem] == kSlot then
                            kFound = 1
                        else
                            kItem = kItem + 1
                        endif
                    od
                    if kFound == 0 then
                        kItem = kCount
                        kFound = 1
                        while kItem > 0 && kFound == 1 do
                            kItem = kItem - 1
                            if kList[kItem] > kSlot then
                                kList[kItem + 1] = kList[kItem]
                            else
                                kItem = kItem + 1
                                kFound = 0
                            endif
                        od
                        kList[kItem] = kSlot
                        kCount = kCount + 1
                    endif
                    kBlockPoolSlot[kBlockCount] = -1
                    kBlockSource[kBlockCount] = kSource[kSlot]
                    kBlockSourceLength[kBlockCount] = kSourceLength[kSlot]
                    kBlockAge[kBlockCount] = kAge[kSlot]
                    kBlockSpan[kBlockCount] = kSpan[kSlot]
                    kBlockReadIndex[kBlockCount] = kRead[kSlot]
                    kBlockStep[kBlockCount] = kStep[kSlot]
                    kBlockLength[kBlockCount] = kLength[kSlot]
                    kBlockSeamWidth[kBlockCount] = kSeamWidth[kSlot]
                    kBlockSeamValue[kBlockCount] = kSeamValue[kSlot]
                    kBlockWarpScale[kBlockCount] = kWarpScale[kSlot]
                    kBlockWarpRadians[kBlockCount] = kWarpRadians[kSlot]
                    kBlockShape[kBlockCount] = kShape[kSlot]
                    kBlockLeft[kBlockCount] = kLeft[kSlot]
                    kBlockRight[kBlockCount] = kRight[kSlot]
                    kBlockCount = kBlockCount + 1
                endif
            endif
            if kTimeJitter > 0 then
                kTimeRandom = (kTimeRandom * 48271) % 2147483647
                kTimeOffset = (2 * kTimeRandom / 2147483647 - 1) * kTimeJitter * 0.45 * sr / kDensity
            else
                kTimeOffset = 0
            endif
            kNextTrigger = max(0, kNextTrigger + sr / kDensity + kTimeOffset - kPreviousTimeOffset)
            kPreviousTimeOffset = kTimeOffset
        endif
        kOffset = kOffset + 1
    od

    aSumL = 0
    aSumR = 0
    kItem = 0
    while kItem < kBlockCount do
        kVoice = kItem
        kPoolSlot = kBlockPoolSlot[kVoice]
        if kPoolSlot >= 0 then
            kAgeNow = kAge[kPoolSlot]
            kReadNow = kRead[kPoolSlot]
            kRateNow = kStep[kPoolSlot]
            kSpanNow = kSpan[kPoolSlot]
            kSourceNow = kSource[kPoolSlot]
            kSourceLengthNow = kSourceLength[kPoolSlot]
            kSeamWidthNow = kSeamWidth[kPoolSlot]
            kSeamValueNow = kSeamValue[kPoolSlot]
            kWarpScaleNow = kWarpScale[kPoolSlot]
            kWarpRadiansNow = kWarpRadians[kPoolSlot]
            kShapeNow = kShape[kPoolSlot]
            kLeftNow = kLeft[kPoolSlot]
            kRightNow = kRight[kPoolSlot]
        else
            kAgeNow = kBlockAge[kVoice]
            kReadNow = kBlockReadIndex[kVoice]
            kRateNow = kBlockStep[kVoice]
            kSpanNow = kBlockSpan[kVoice]
            kSourceNow = kBlockSource[kVoice]
            kSourceLengthNow = kBlockSourceLength[kVoice]
            kSeamWidthNow = kBlockSeamWidth[kVoice]
            kSeamValueNow = kBlockSeamValue[kVoice]
            kWarpScaleNow = kBlockWarpScale[kVoice]
            kWarpRadiansNow = kBlockWarpRadians[kVoice]
            kShapeNow = kBlockShape[kVoice]
            kLeftNow = kBlockLeft[kVoice]
            kRightNow = kBlockRight[kVoice]
        endif
        ; Phase is clamped, so samples before the start or after the end are silent.
        aPhase limit (aRamp + kAgeNow) / kSpanNow, 0, 1
        if kShapeNow == 0 then
            aEnvelope = $CM_HANN(aPhase)
        elseif kShapeNow == 1 then
            aEnvelope = $CM_TRIANGLE(aPhase)
        else
            aEnvelope = $CM_BLACKMAN(aPhase)
        endif
        ; Pitch-down/equal-rate playback uses cubic interpolation. Pitch-up uses
        ; Csound's sinc table reader with a cutoff warped to the playback step;
        ; select an init-time window large enough for 8 samples per step.
        aIndex = kReadNow + kRateNow * aRamp
        kAbsRateNow = abs(kRateNow)
        if iMode == 2 && kWarpScaleNow > 0 then
            ; φ' = φ + β sin(2πφ)/(2π), 0 ≤ β ≤ 1. The warp keeps each
            ; cycle boundary fixed, moves waveform transitions within the cycle,
            ; and never reverses phase.
            aIndex = aIndex + kWarpScaleNow * sin(kWarpRadiansNow * aIndex)
        endif
        if iMode == 2 then
            ; Oscillator tables are already band-limited by their harmonic bank;
            ; their dense one-period table index step is not an audio sample rate.
            aSample tablexkt aIndex, kSourceNow, 0, 4, 0, 0, 1
        elseif kSourceLengthNow < 16 then
            ; Very short tables cannot support the sinc window; keep their wrapped read finite.
            aSample tablexkt aIndex, kSourceNow, 0, 2, 0, 0, 1
        elseif kAbsRateNow <= 1 then
            aSample tablexkt aIndex, kSourceNow, 0, 4, 0, 0, 1
        elseif kAbsRateNow <= 2 then
            aSample tablexkt aIndex, kSourceNow, kAbsRateNow, 16, 0, 0, 1
        elseif kAbsRateNow <= 4 then
            aSample tablexkt aIndex, kSourceNow, kAbsRateNow, 32, 0, 0, 1
        elseif kAbsRateNow <= 8 then
            aSample tablexkt aIndex, kSourceNow, kAbsRateNow, 64, 0, 0, 1
        elseif kAbsRateNow <= 16 then
            aSample tablexkt aIndex, kSourceNow, kAbsRateNow, 128, 0, 0, 1
        elseif kAbsRateNow <= 64 then
            aSample tablexkt aIndex, kSourceNow, kAbsRateNow, 512, 0, 0, 1
        else
            aSample tablexkt aIndex, kSourceNow, kAbsRateNow, 1024, 0, 0, 1
        endif
        if iMode == 0 && kSeamWidthNow > 0 then
            kFileLength = kSourceLengthNow
            aWrapped = aIndex - floor(aIndex / kFileLength) * kFileLength
            aSeamDistance = kFileLength * 0.5 - abs(aWrapped - kFileLength * 0.5)
            aSeamBlend limit 1 - aSeamDistance / kSeamWidthNow, 0, 1
            aSeamBlend = aSeamBlend * aSeamBlend * (3 - 2 * aSeamBlend)
            aSample = aSample + (kSeamValueNow - aSample) * aSeamBlend
        endif
        aVoice = aSample * aEnvelope
        aSumL = aSumL + aVoice * kLeftNow
        aSumR = aSumR + aVoice * kRightNow
        kItem = kItem + 1
    od

    ; Advance reusable state only after all snapshots have been rendered. A slot
    ; reused inside the block now holds the new grain; old block tails stay intact.
    kBlockCount = kCount
    kItem = 0
    kCount = 0
    while kItem < kBlockCount do
        kVoice = kList[kItem]
        if kLength[kVoice] > 0 then
            if kAge[kVoice] + iBlock >= kLength[kVoice] then
                kLength[kVoice] = 0
            else
                kAge[kVoice] = kAge[kVoice] + iBlock
                kNewRead = kRead[kVoice] + kStep[kVoice] * iBlock
                kRead[kVoice] = kNewRead - floor(kNewRead / kSourceLength[kVoice]) * kSourceLength[kVoice]
                kList[kCount] = kVoice
                kCount = kCount + 1
            endif
        endif
        kItem = kItem + 1
    od
    if iMode == 2 then
        kCarrier = kCarrier + kCarrierStep * iBlock
        kCarrier = kCarrier - floor(kCarrier / kFrames) * kFrames
    endif

    if iMode == 1 && kFreeze < 0.5 then
        kWrite = (kWrite + iBlock) % kFrames
        kFilled = min(kFilled + iBlock, kFrames)
    endif

    ; Dashboard frame: header 0..15, source overview 16..207, then 64
    ; triples (source position, window progress, current amplitude). A negative position
    ; identifies a voice still playing a different source after a bank change.
    if iDisplay > 0 then
        ; Scan the loaded GEN01 table incrementally. Bound work per audio block,
        ; including on long files. Only completed pixels are published.
        kSampleSource init -1
        kScanFrame init 0
        kOverviewBins init 0
        kMinimum init 0
        kMaximum init 0
        if iMode == 0 then
            if kSampleSource != kTable then
                kSampleSource = kTable
                kScanFrame = 0
                kOverviewBins = 0
                kMinimum = 0
                kMaximum = 0
            endif
            kEnd = min(kFrames, kScanFrame + 128)
            while kScanFrame < kEnd do
                kSampleValue tablekt kScanFrame, kTable
                kMinimum = min(kMinimum, kSampleValue)
                kMaximum = max(kMaximum, kSampleValue)
                kScanFrame += 1
                while kOverviewBins < 192 && kScanFrame * 192 >= (kOverviewBins + 1) * kFrames do
                    tablewkt kMaximum, 16 + kOverviewBins, iDisplay
                    tablewkt kMinimum, 465 + kOverviewBins, iDisplay
                    kOverviewBins += 1
                    if kScanFrame * 192 < (kOverviewBins + 1) * kFrames then
                        kMinimum = 0
                        kMaximum = 0
                    endif
                od
            od
        endif
        kFrameTick metro 30
        if kFrameTick == 1 then
            kView = iDisplay
            kHasSource = (iMode == 1 && kSnapshotSelect == 1 && kSnapshotActive == 0 ? 0 : 1)
            kWriter = (kSnapshotSelect == 1 ? kSnapshotWrite : kWrite) / kFrames
            kRecorded = (kSnapshotSelect == 1 ? kSnapshotFilled : kFilled) / kFrames
            tablewkt 1, 0, kView
            tablewkt kCount, 1, kView
            tablewkt kHeldPosition, 2, kView
            tablewkt kPosJitter, 3, kView
            tablewkt kWriter, 4, kView
            tablewkt kRecorded, 5, kView
            tablewkt kHasSource, 6, kView
            tablewkt int(limit(kWindow, 0, 2) + 0.5), 7, kView
            tablewkt kReverse, 8, kView
            tablewkt kFreeze, 9, kView
            tablewkt kPitch, 10, kView
            tablewkt kSnapshotSelect, 11, kView
            tablewkt sr, 12, kView
            tablewkt kCarrier / kFrames, 13, kView
            tablewkt kOverviewBins, 14, kView
            tablewkt kSpread, 15, kView
            tablewkt kFrames / kSourceRate, 657, kView
            tablewkt kSourceRate, 658, kView
            kPixel = 0
            while iMode != 0 && kPixel < 192 do
                if iMode == 2 then
                    kAngle = kPixel / 192 * 6.283185307179586
                    kPhase = (kAngle + kHeldPosition * sin(kAngle)) / 6.283185307179586
                    kValue tablekt kPhase * kFrames, kTable
                elseif kSnapshotSelect == 1 then
                    kValue = kCapturedPeak[kPixel]
                else
                    kValue = kRingPeak[kPixel]
                endif
                tablewkt kValue * kHasSource, 16 + kPixel, kView
                kPixel += 1
            od
            kEnvelopeShape = int(limit(kWindow, 0, 2) + 0.5)
            kPoint = 0
            while kPoint < 65 do
                kPhaseView = kPoint / 64
                if kEnvelopeShape == 0 then
                    kAmplitude = $CM_HANN(kPhaseView)
                elseif kEnvelopeShape == 1 then
                    kAmplitude = $CM_TRIANGLE(kPhaseView)
                else
                    kAmplitude = $CM_BLACKMAN(kPhaseView)
                endif
                tablewkt kAmplitude, 400 + kPoint, kView
                kPoint += 1
            od
            kItem = 0
            while kItem < kCount do
                kVoice = kList[kItem]
                kOffsetView = 208 + 3 * kItem
                kPosView = (kSource[kVoice] == kReadTable ? kRead[kVoice] / kSourceLength[kVoice] : -1)
                tablewkt kPosView, kOffsetView, kView
                tablewkt kAge[kVoice] / kSpan[kVoice], kOffsetView + 1, kView
                kPhaseView = limit(kAge[kVoice] / kSpan[kVoice], 0, 1)
                if kShape[kVoice] == 0 then
                    kAmplitude = $CM_HANN(kPhaseView)
                elseif kShape[kVoice] == 1 then
                    kAmplitude = $CM_TRIANGLE(kPhaseView)
                else
                    kAmplitude = $CM_BLACKMAN(kPhaseView)
                endif
                tablewkt kAmplitude, kOffsetView + 2, kView
                kItem += 1
            od
        endif
    endif

    ; Smooth actual voice count over 10 ms. Module wrappers apply a hard safety ceiling
    ; only after the user-selected output gain; the default core path stays linear.
    kSmoothedActive = kSmoothedActive + (max(1, kCount) - kSmoothedActive) * (1 - exp(-iBlock / (0.01 * sr)))
    kGain = 1 / sqrt(max(1, kSmoothedActive))
    aLeft = aSumL * kGain
    aRight = aSumR * kGain
    if iMode == 1 && kFeedback > 0 then
        kFeedbackPrevious downsamp (aLeft + aRight) * 0.5
    else
        kFeedbackPrevious = 0
    endif
    xout aLeft, aRight
endop
