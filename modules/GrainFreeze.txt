;@ requires GrainCore.txt
opcode GrainFreeze, 0, kkkkkkkkkkkkkkkkkkkkkkkkkkkkkkkkkkkko
    kDensity, kSize, kPosition, kPosJitter, kPitch, kPitchJitter, kWindow, kPan, kSpread, kFreeze, kMix, kLevel, kSeed, kTimeJitter, kReverse, kSource, kFeedback, kCapture, kAudioIn, kDensityIn, kSizeIn, kPositionIn, kPosJitterIn, kPitchIn, kPitchJitterIn, kWindowIn, kPanIn, kSpreadIn, kFreezeIn, kMixIn, kLevelIn, kTimeJitterIn, kReverseIn, kFeedbackIn, kOutL, kOutR, iDisplay xin

    aDry zar kAudioIn
    kDensityCV zkr kDensityIn
    kDensity = kDensity + kDensityCV * 24
    kSizeCV zkr kSizeIn
    kSize = kSize + kSizeCV * 200
    kPositionCV zkr kPositionIn
    kPosition = kPosition / 127 + kPositionCV * 1
    kPosJitterCV zkr kPosJitterIn
    kPosJitter = kPosJitter / 127 + kPosJitterCV * 1
    kPitchCV zkr kPitchIn
    kPitch = kPitch + kPitchCV * 12
    kPitchJitterCV zkr kPitchJitterIn
    kPitchJitter = kPitchJitter + kPitchJitterCV * 12
    kWindowCV zkr kWindowIn
    kWindow = kWindow + kWindowCV * 2
    kPanCV zkr kPanIn
    kPan = kPan / 127 + kPanCV * 1
    kSpreadCV zkr kSpreadIn
    kSpread = kSpread / 127 + kSpreadCV * 1
    kFreezeCV zkr kFreezeIn
    kFreeze = kFreeze + kFreezeCV * 1
    kMixCV zkr kMixIn
    kMix = kMix / 127 + kMixCV * 1
    kLevelCV zkr kLevelIn
    kLevel = kLevel + kLevelCV * 12
    kTimeJitterCV zkr kTimeJitterIn
    kTimeJitter = kTimeJitter / 100 + kTimeJitterCV * 1
    kReverseCV zkr kReverseIn
    kReverse = kReverse + kReverseCV * 1
    kFeedbackCV zkr kFeedbackIn
    kFeedback = kFeedback + kFeedbackCV * 0.25
    kDensity limit kDensity, 0, 24
    kSize limit kSize, 2, 400
    iRingFrames = int(8 * sr)
    iRing ftgentmp 0, 0, -iRingFrames, -2, 0
    iSnapshotA ftgentmp 0, 0, -iRingFrames, -2, 0
    iSnapshotB ftgentmp 0, 0, -iRingFrames, -2, 0
    kTable = iRing
    kSnapshotA = iSnapshotA
    kSnapshotB = iSnapshotB
    kFrames = iRingFrames
    kRate = sr

    kLevel limit kLevel, -24, 12
    kLevel = pow(10, kLevel / 20)
    aLevel interp kLevel
    kMix limit kMix, 0, 1
    aMix interp kMix
    aOutL limit ((1 - aMix) * aDry + aMix * aWetL) * aLevel, -1, 1
    aOutR limit ((1 - aMix) * aDry + aMix * aWetR) * aLevel, -1, 1
    if iDisplay > 0 then
        CMPulvisOutputView aOutL, iDisplay
    endif
    if kOutL > 1 then
        zaw aOutL, kOutL
    endif
    if kOutR > 1 then
        zaw aOutR, kOutR
    endif
endop
